## Understanding Authentication

### Sonamu's Authentication Philosophy

Sonamu **does not force a specific authentication library**. Instead, it only provides an interface for injecting authenticated user information into Context. This has the following advantages:

1. **Flexibility**: Use any authentication system you want - Passport.js, JWT, Auth0, Firebase Auth, etc.
2. **Independence**: Authentication logic is separated from business logic
3. **Testability**: Test by injecting fake users into Context

### Authentication Flow

```
1. Client → Server (Login request)
2. Server → Authentication verification (Passport/JWT etc.)
3. Server → Inject user info into Context
4. API method → Access Sonamu.getContext().user
5. Guards → Permission verification (admin, user, etc.)
```

## User Type Definition

### Basic User Interface

First, define the User type to be used in your application.

```typescript
// types/user.ts
export interface User {
  id: number;
  email: string;
  username: string;
  role: "admin" | "user" | "guest";
  createdAt: Date;
  updatedAt: Date;
}

// Represent unauthenticated state
export type AuthUser = User | null;
```

### Context Type Extension

Add User type to Sonamu's Context.

```typescript
// types/context.ts
import type { BaseContext } from "sonamu";
import type { User } from "./user";

export interface AppContext extends BaseContext {
  user: User | null;
}
```

<Info>
Context is created for each request and contains information accessible throughout the request.
</Info>

## Passport.js Integration

Passport.js is the most widely used authentication middleware in Node.js. Learn how to use it with Sonamu.

### Install Passport

```bash
npm install passport passport-local
npm install -D @types/passport @types/passport-local
# or
pnpm add passport passport-local
pnpm add -D @types/passport @types/passport-local
```

### Passport Strategy Configuration

Set up the local authentication strategy. This is the most basic method of authenticating with username and password.

```typescript
// auth/passport.config.ts
import passport from "passport";
import { Strategy as LocalStrategy } from "passport-local";
import bcrypt from "bcrypt";
import type { User } from "../types/user";

/**
 * Local authentication strategy setup
 * 
 * usernameField and passwordField are the field names sent by the client.
 * Defaults are 'username' and 'password', but can be changed to 'email' and 'password'.
 */
passport.use(
  new LocalStrategy(
    {
      usernameField: "email", // Login with email
      passwordField: "password",
    },
    async (email, password, done) => {
      try {
        // Query user from database
        const db = getDatabase(); // Use your project's DB access method
        const user = await db
          .table("users")
          .where("email", email)
          .first();
        
        if (!user) {
          // User not found
          return done(null, false, { message: "Incorrect email" });
        }
        
        // Verify password
        const isPasswordValid = await bcrypt.compare(password, user.password);
        
        if (!isPasswordValid) {
          // Password mismatch
          return done(null, false, { message: "Incorrect password" });
        }
        
        // Authentication successful - remove password field and return
        const { password: _, ...userWithoutPassword } = user;
        return done(null, userWithoutPassword as User);
      } catch (error) {
        return done(error);
      }
    }
  )
);

/**
 * Save user info to session
 * 
 * serializeUser is called after successful login and determines what to store in the session.
 * Usually only the user ID is stored to minimize session size.
 */
passport.serializeUser((user: any, done) => {
  done(null, user.id);
});

/**
 * Restore user info from session
 * 
 * deserializeUser is called on every request and queries user info using the ID stored in session.
 * The info queried here is stored in req.user.
 */
passport.deserializeUser(async (id: number, done) => {
  try {
    const db = getDatabase();
    const user = await db
      .table("users")
      .where("id", id)
      .first();
    
    if (!user) {
      return done(new Error("User not found"));
    }
    
    const { password: _, ...userWithoutPassword } = user;
    done(null, userWithoutPassword as User);
  } catch (error) {
    done(error);
  }
});

export default passport;
```

### Integration with Fastify Server

Connect Passport to the Fastify server.

```typescript
// server.ts
import fastify from "fastify";
import fastifySession from "@fastify/session";
import fastifyCookie from "@fastify/cookie";
import passport from "./auth/passport.config";

const app = fastify();

/**
 * Register cookie parser
 * Cookies are required to use sessions.
 */
app.register(fastifyCookie);

/**
 * Register session middleware
 * 
 * secret: Key to encrypt session cookie (must be managed via environment variable)
 * cookie: Cookie settings
 *   - secure: Only send cookie over HTTPS (true in production)
 *   - httpOnly: Cookie inaccessible from JavaScript (XSS prevention)
 *   - maxAge: Cookie validity period (milliseconds)
 */
app.register(fastifySession, {
  secret: process.env.SESSION_SECRET!, // Read from environment variable
  cookie: {
    secure: process.env.NODE_ENV === "production", // HTTPS required
    httpOnly: true, // XSS attack prevention
    maxAge: 1000 * 60 * 60 * 24 * 7, // 7 days
  },
  saveUninitialized: false, // Don't save uninitialized sessions
  resave: false, // Don't resave unchanged sessions
});

/**
 * Initialize Passport
 * 
 * passport.initialize() connects Passport to Fastify.
 * passport.session() enables session-based authentication.
 */
app.addHook("onRequest", (request, reply, done) => {
  passport.initialize()(request.raw, reply.raw, done);
});

app.addHook("onRequest", (request, reply, done) => {
  passport.session()(request.raw, reply.raw, done);
});

// Register API routes
// ...
```

<Warning>
**Security Notes**:
- Never hardcode `SESSION_SECRET` in code
- Must set `secure: true` in production (HTTPS required)
- Prevent XSS attacks with `httpOnly: true`
- Recommended to use external storage like Redis for session store (maintains sessions on server restart)
</Warning>

## JWT Authentication

You can also implement authentication using JWT (JSON Web Token) instead of sessions. JWT is a stateless authentication method, useful in microservice environments.

### Install JWT Library

```bash
npm install jsonwebtoken
npm install -D @types/jsonwebtoken
# or
pnpm add jsonwebtoken
pnpm add -D @types/jsonwebtoken
```

### JWT Utilities

Create functions for JWT token generation and verification.

```typescript
// auth/jwt.ts
import jwt from "jsonwebtoken";
import type { User } from "../types/user";

const JWT_SECRET = process.env.JWT_SECRET!;
const JWT_EXPIRES_IN = "7d"; // 7 days

/**
 * JWT payload type
 * 
 * Include only minimal information to store in JWT.
 * Never include sensitive information (passwords, card numbers, etc.).
 */
export interface JWTPayload {
  userId: number;
  email: string;
  role: string;
}

/**
 * Generate JWT token
 * 
 * @param user User information
 * @returns JWT token string
 */
export function generateToken(user: User): string {
  const payload: JWTPayload = {
    userId: user.id,
    email: user.email,
    role: user.role,
  };
  
  return jwt.sign(payload, JWT_SECRET, {
    expiresIn: JWT_EXPIRES_IN,
  });
}

/**
 * Verify JWT token
 * 
 * @param token JWT token
 * @returns Verified payload or null
 */
export function verifyToken(token: string): JWTPayload | null {
  try {
    const payload = jwt.verify(token, JWT_SECRET) as JWTPayload;
    return payload;
  } catch (error) {
    // Token is invalid or expired
    console.error("JWT verification failed:", error);
    return null;
  }
}

/**
 * Generate Refresh Token
 * 
 * A token with longer validity than Access Token.
 * When Access Token expires, use Refresh Token to get a new Access Token.
 */
export function generateRefreshToken(user: User): string {
  const payload: JWTPayload = {
    userId: user.id,
    email: user.email,
    role: user.role,
  };
  
  return jwt.sign(payload, JWT_SECRET, {
    expiresIn: "30d", // 30 days
  });
}
```

### JWT Middleware

Create middleware to extract and verify JWT from request headers.

```typescript
// auth/jwt.middleware.ts
import type { FastifyRequest, FastifyReply } from "fastify";
import { verifyToken } from "./jwt";

/**
 * JWT authentication middleware
 * 
 * Extracts and verifies Bearer token from Authorization header.
 * On successful verification, queries user info from database and adds to request.
 */
export async function jwtAuthMiddleware(
  request: FastifyRequest,
  reply: FastifyReply
) {
  // Check Authorization header
  const authHeader = request.headers.authorization;
  
  if (!authHeader || !authHeader.startsWith("Bearer ")) {
    // No JWT token, just pass through (optional auth)
    // Will be blocked by guards if needed
    return;
  }
  
  // Remove Bearer prefix
  const token = authHeader.substring(7);
  
  // Verify token
  const payload = verifyToken(token);
  
  if (!payload) {
    // Token is invalid
    return reply.status(401).send({
      error: "Invalid or expired token",
    });
  }
  
  // Query user from database
  const db = getDatabase();
  const user = await db
    .table("users")
    .where("id", payload.userId)
    .first();
  
  if (!user) {
    return reply.status(401).send({
      error: "User not found",
    });
  }
  
  // Add user info to request
  // Accessible from Context later
  request.user = user;
}
```

## Injecting User into Context

Here's how to inject authenticated user information into Sonamu Context. This is **the most important step**, enabling access to user info via `Sonamu.getContext().user` in API methods.

### Add User When Creating Context

```typescript
// server.ts
import { Sonamu } from "sonamu";
import type { AppContext } from "./types/context";

app.addHook("onRequest", async (request, reply) => {
  // When using Passport: user info is in request.user
  // When using JWT: jwtAuthMiddleware sets request.user
  
  const context: AppContext = {
    request,
    reply,
    user: request.user || null, // Authenticated user or null
  };
  
  // Set Sonamu Context
  Sonamu.setContext(context);
});
```

### Accessing User Info in API

Now you can access user info in all API methods.

```typescript
class UserModel extends BaseModelClass {
  @api({ httpMethod: "GET" })
  async getProfile(): Promise<{
    user: User;
  }> {
    // Get user info from Context
    const context = Sonamu.getContext();
    
    if (!context.user) {
      throw new Error("Authentication required");
    }
    
    return {
      user: context.user,
    };
  }
  
  @api({ httpMethod: "PUT" })
  async updateProfile(params: {
    username?: string;
    bio?: string;
  }): Promise<{ user: User }> {
    const context = Sonamu.getContext();
    
    if (!context.user) {
      throw new Error("Authentication required");
    }
    
    const { username, bio } = params;
    
    // Can only modify own profile
    const wdb = this.getPuri("w");
    await wdb
      .table("users")
      .where("id", context.user.id)
      .update({
        username,
        bio,
        updated_at: new Date(),
      });
    
    // Query updated user info
    const rdb = this.getPuri("r");
    const updatedUser = await rdb
      .table("users")
      .where("id", context.user.id)
      .first();
    
    return { user: updatedUser };
  }
}
```

## Login/Logout API

### Passport Login

```typescript
class AuthModel extends BaseModelClass {
  @api({ httpMethod: "POST" })
  async login(params: {
    email: string;
    password: string;
  }): Promise<{
    success: boolean;
    user: User;
  }> {
    const { email, password } = params;
    const context = Sonamu.getContext();
    
    // Passport authentication
    return new Promise((resolve, reject) => {
      passport.authenticate("local", (err, user, info) => {
        if (err) {
          reject(err);
          return;
        }
        
        if (!user) {
          reject(new Error(info?.message || "Login failed"));
          return;
        }
        
        // Save login info to session
        context.request.logIn(user, (err) => {
          if (err) {
            reject(err);
            return;
          }
          
          resolve({
            success: true,
            user,
          });
        });
      })(context.request.raw, context.reply.raw, () => {});
    });
  }
  
  @api({ httpMethod: "POST" })
  async logout(): Promise<{ success: boolean }> {
    const context = Sonamu.getContext();
    
    if (!context.user) {
      throw new Error("Not logged in");
    }
    
    return new Promise((resolve, reject) => {
      context.request.logOut((err) => {
        if (err) {
          reject(err);
          return;
        }
        
        resolve({ success: true });
      });
    });
  }
}
```

### JWT Login

```typescript
import { generateToken, generateRefreshToken } from "./auth/jwt";
import bcrypt from "bcrypt";

class AuthModel extends BaseModelClass {
  @api({ httpMethod: "POST" })
  async login(params: {
    email: string;
    password: string;
  }): Promise<{
    accessToken: string;
    refreshToken: string;
    user: User;
  }> {
    const { email, password } = params;
    
    // Query user
    const rdb = this.getPuri("r");
    const user = await rdb
      .table("users")
      .where("email", email)
      .first();
    
    if (!user) {
      throw new Error("Invalid email or password");
    }
    
    // Verify password
    const isPasswordValid = await bcrypt.compare(password, user.password);
    
    if (!isPasswordValid) {
      throw new Error("Invalid email or password");
    }
    
    // Generate JWT tokens
    const accessToken = generateToken(user);
    const refreshToken = generateRefreshToken(user);
    
    // Save Refresh Token to database (optional)
    const wdb = this.getPuri("w");
    await wdb.table("refresh_tokens").insert({
      user_id: user.id,
      token: refreshToken,
      expires_at: new Date(Date.now() + 30 * 24 * 60 * 60 * 1000), // 30 days
    });
    
    const { password: _, ...userWithoutPassword } = user;
    
    return {
      accessToken,
      refreshToken,
      user: userWithoutPassword,
    };
  }
  
  @api({ httpMethod: "POST" })
  async refreshAccessToken(params: {
    refreshToken: string;
  }): Promise<{
    accessToken: string;
  }> {
    const { refreshToken } = params;
    
    // Verify Refresh Token
    const payload = verifyToken(refreshToken);
    
    if (!payload) {
      throw new Error("Invalid refresh token");
    }
    
    // Verify Refresh Token in database
    const rdb = this.getPuri("r");
    const storedToken = await rdb
      .table("refresh_tokens")
      .where("token", refreshToken)
      .where("user_id", payload.userId)
      .first();
    
    if (!storedToken) {
      throw new Error("Refresh token not found");
    }
    
    // Query user
    const user = await rdb
      .table("users")
      .where("id", payload.userId)
      .first();
    
    if (!user) {
      throw new Error("User not found");
    }
    
    // Generate new Access Token
    const accessToken = generateToken(user);
    
    return { accessToken };
  }
}
```
